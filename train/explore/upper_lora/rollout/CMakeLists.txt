# 판단 모델 결과 라벨용 갈래 굴리기 (dec_rollout.cpp) — 시뮬(train/explore/sim)을 하위 프로젝트로 받아 limo_sim 에 링크 (sim CMakeLists 는 안 건드림)
cmake_minimum_required(VERSION 3.22)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES AND NOT DEFINED ENV{CUDAARCHS})
  set(CMAKE_CUDA_ARCHITECTURES 86)
endif()
project(dec_rollout LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(EXPLORE ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_subdirectory(${EXPLORE}/sim ${CMAKE_BINARY_DIR}/sim EXCLUDE_FROM_ALL)
find_package(ZLIB REQUIRED)
add_executable(dec_rollout dec_rollout.cpp ${EXPLORE}/scripts/decider.cpp)
target_include_directories(dec_rollout PRIVATE ${EXPLORE}/scripts ${EXPLORE}/../../src/limo_fire/limo_fire_explore/include)   # limo_fire_explore/decider_io.h (판단 모델 요청 · 답)
target_link_libraries(dec_rollout limo_sim ZLIB::ZLIB)
