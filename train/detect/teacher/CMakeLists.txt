cmake_minimum_required(VERSION 3.22)
project(sam3_teacher LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 20)
set(CMAKE_CUDA_ARCHITECTURES 120)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(CUDAToolkit REQUIRED)
find_package(nlohmann_json REQUIRED)
set(CMAKE_CUDA_FLAGS "${CMAKE_CUDA_FLAGS} --expt-relaxed-constexpr -lineinfo")

add_library(sam3 STATIC src/prep.cpp src/st.cpp src/tok.cpp src/k.cu src/fa.cu src/sam3.cu)
target_include_directories(sam3 PUBLIC src third_party)
target_link_libraries(sam3 PUBLIC CUDA::cudart CUDA::cublas CUDA::cublasLt nlohmann_json::nlohmann_json)

add_executable(sam3_parity tools/sam3_parity.cpp)
target_link_libraries(sam3_parity sam3)
add_executable(sam3_bench tools/sam3_bench.cpp)
target_link_libraries(sam3_bench sam3)
add_executable(sam3_label tools/sam3_label.cpp)
target_link_libraries(sam3_label sam3)
