cmake_minimum_required(VERSION 3.22)
project(yolo26_trainer LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 20)
set(CMAKE_CUDA_ARCHITECTURES 120)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(CUDAToolkit REQUIRED)
find_package(nlohmann_json REQUIRED)
set(CMAKE_CUDA_FLAGS "${CMAKE_CUDA_FLAGS} --expt-relaxed-constexpr --extended-lambda -lineinfo")

set(NET_SRC src/st.cpp src/conv.cpp src/kern.cu src/bnv.cu src/attn.cu src/net.cpp src/loss.cu src/segloss.cu src/optim.cu src/cache.cpp src/data.cpp src/augment.cu src/evalmap.cpp src/evalmask.cu)
# 학습용(fp16)과 정합성 시험용(fp32) 두 벌
add_library(ytnet STATIC ${NET_SRC})
add_library(ytnet32 STATIC ${NET_SRC})
target_compile_definitions(ytnet32 PUBLIC YT_FP32)
foreach(t ytnet ytnet32)
  # det_metrics.hpp (짝짓기·AP): 인식 벤치마크와 같이 쓰는 헤더
  target_include_directories(${t} PUBLIC src third_party/cudnn_frontend/include
                             ${CMAKE_CURRENT_SOURCE_DIR}/../../../src/limo_fire/limo_fire_perception/include)
  set_source_files_properties(src/net.cpp src/conv.cpp src/data.cpp src/cache.cpp src/evalmap.cpp PROPERTIES LANGUAGE CUDA)
  target_link_libraries(${t} PUBLIC CUDA::cudart CUDA::cublas cudnn turbojpeg nlohmann_json::nlohmann_json CUDA::nvrtc CUDA::cuda_driver)
endforeach()

add_executable(net_parity tools/net_parity.cpp)
target_link_libraries(net_parity ytnet32)
add_executable(loss_parity tools/loss_parity.cpp)
target_link_libraries(loss_parity ytnet32)
set_source_files_properties(tools/loss_parity.cpp PROPERTIES LANGUAGE CUDA)
add_executable(optim_parity tools/optim_parity.cpp)
target_link_libraries(optim_parity ytnet32)
set_source_files_properties(tools/optim_parity.cpp PROPERTIES LANGUAGE CUDA)
add_executable(build_cache tools/build_cache.cpp src/cache.cpp)
target_include_directories(build_cache PRIVATE src)
target_link_libraries(build_cache CUDA::cudart turbojpeg)
add_executable(train tools/train.cpp)
target_link_libraries(train ytnet)
set_source_files_properties(tools/train.cpp PROPERTIES LANGUAGE CUDA)
add_executable(aug_dump tools/aug_dump.cpp)
target_link_libraries(aug_dump ytnet)
set_source_files_properties(tools/aug_dump.cpp PROPERTIES LANGUAGE CUDA)
add_executable(op_time tools/op_time.cpp)
target_link_libraries(op_time ytnet)
set_source_files_properties(tools/op_time.cpp PROPERTIES LANGUAGE CUDA)
