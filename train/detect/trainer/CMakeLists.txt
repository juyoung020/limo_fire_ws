cmake_minimum_required(VERSION 3.22)
project(yolo26_trainer LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 20)
set(CMAKE_CUDA_ARCHITECTURES 120)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(CUDAToolkit REQUIRED)
find_package(nlohmann_json REQUIRED)
set(CMAKE_CUDA_FLAGS "${CMAKE_CUDA_FLAGS} --expt-relaxed-constexpr --extended-lambda -lineinfo")

set(NET_SRC src/st.cpp src/conv.cpp src/kern.cu src/attn.cu src/net.cpp src/loss.cu)
# 학습용(fp16)과 정합성 시험용(fp32) 두 벌
add_library(ytnet STATIC ${NET_SRC})
add_library(ytnet32 STATIC ${NET_SRC})
target_compile_definitions(ytnet32 PUBLIC YT_FP32)
foreach(t ytnet ytnet32)
  target_include_directories(${t} PUBLIC src)
  set_source_files_properties(src/net.cpp src/conv.cpp PROPERTIES LANGUAGE CUDA)
  target_link_libraries(${t} PUBLIC CUDA::cudart CUDA::cublas cudnn nlohmann_json::nlohmann_json)
endforeach()

add_executable(net_parity tools/net_parity.cpp)
target_link_libraries(net_parity ytnet32)
add_executable(loss_parity tools/loss_parity.cpp)
target_link_libraries(loss_parity ytnet32)
set_source_files_properties(tools/loss_parity.cpp PROPERTIES LANGUAGE CUDA)
