# sensor_view — ESP32 시험 코드(firmware/sensor_check) 화면. ROS 아님 (tools/ 는 COLCON_IGNORE)
cmake_minimum_required(VERSION 3.16)
project(sensor_view LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_executable(sensor_view src/main.cpp)
target_compile_options(sensor_view PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(sensor_view Qt5::Widgets)
