cmake_minimum_required(VERSION 3.8)
project(limo_fire_bt)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
set(deps lifecycle_msgs rclcpp rclcpp_action rclcpp_lifecycle nav2_msgs nav2_costmap_2d nav_msgs sensor_msgs geometry_msgs std_msgs tf2_ros tf2_geometry_msgs pluginlib limo_fire_msgs)
foreach(dep ${deps})
 find_package(${dep} REQUIRED)
endforeach()
foreach(exe fire_manager mock_extinguisher fire_demo twist_stamper)
 add_executable(${exe} src/${exe}.cpp)
 target_include_directories(${exe} PRIVATE include)
 ament_target_dependencies(${exe} ${deps})
 install(TARGETS ${exe} DESTINATION lib/${PROJECT_NAME})
endforeach()
add_library(danger_layer SHARED src/danger_layer.cpp)
ament_target_dependencies(danger_layer ${deps})
pluginlib_export_plugin_description_file(nav2_costmap_2d danger_plugins.xml)
install(TARGETS danger_layer LIBRARY DESTINATION lib)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
# Jazzy registers Nav2 built-in BT nodes itself. Generate from the single base
# config instead of maintaining a second complete copy or passing an untyped [].
set(nav2_base "${CMAKE_CURRENT_SOURCE_DIR}/../limo_fire_navigation/config/nav2_params.yaml")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${nav2_base}")
file(READ "${nav2_base}" nav2_jazzy)
string(REGEX REPLACE "    plugin_lib_names:\n(      - [^\n]*\n)*" "" nav2_jazzy "${nav2_jazzy}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/nav2_base_jazzy.yaml" "${nav2_jazzy}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/nav2_base_jazzy.yaml" DESTINATION share/${PROJECT_NAME}/config)
if(BUILD_TESTING)
 find_package(rosgraph_msgs REQUIRED)
 enable_testing()
 add_executable(danger_layer_test test/danger_layer_test.cpp)
 ament_target_dependencies(danger_layer_test ${deps})
 add_test(NAME danger_layer_test COMMAND danger_layer_test)
 set_tests_properties(danger_layer_test PROPERTIES TIMEOUT 10 RUN_SERIAL TRUE ENVIRONMENT "ROS_DOMAIN_ID=175")
 add_executable(protocol_test test/protocol_test.cpp)
 target_include_directories(protocol_test PRIVATE include)
 ament_target_dependencies(protocol_test ${deps} rosgraph_msgs)
 set(protocol_domain 180)
 foreach(scenario small_success ext_failure large_success no_path bad_input depth_missing duplicate nav_failure ext_timeout)
  add_test(NAME protocol_${scenario} COMMAND protocol_test --ros-args --params-file ${CMAKE_CURRENT_SOURCE_DIR}/config/fire_manager.yaml -p use_sim_time:=true -p scenario:=${scenario} -p phase_timeout_s:=5.0)
  set_tests_properties(protocol_${scenario} PROPERTIES TIMEOUT 20 RUN_SERIAL TRUE ENVIRONMENT "ROS_DOMAIN_ID=${protocol_domain}")
  math(EXPR protocol_domain "${protocol_domain}+1")
 endforeach()
 add_executable(safety_test test/safety_test.cpp)
 target_include_directories(safety_test PRIVATE include)
 target_compile_options(safety_test PRIVATE -UNDEBUG)
 target_compile_options(danger_layer_test PRIVATE -UNDEBUG)
 add_test(NAME safety_test COMMAND safety_test)
endif()
ament_package()
